# =============================================================================
# 실습: 디스크 생성 + 마운트 (통합)
# -----------------------------------------------------------------------------
# CycleCloud 템플릿에서 디스크는 "volume 블록(생성/attach)" 과
# "mount 설정(포맷/마운트)" 이 한 쌍으로 동작합니다. 이 파일 하나로
# 생성 + 마운트를 모두 처리합니다.
#
# 붙여넣는 위치:
#   slurm.txt 의  [[node nodearraybase]]  블록 하위 —
#   **일반 [[[configuration]]] 속성(slurm.node_prefix 등) 뒤, 첫 [[[cluster-init 앞]]]**.
#   (기존 [[[volume boot]]] 블록과 같은 들여쓰기 레벨. 모든 execute 노드에 공통 적용.
#    특정 파티션만 원하면 해당 [[nodearray ...]] 블록 안에 넣으세요.)
#
#   ⚠️ [[[configuration]]] 헤더 '바로 뒤'에 volume/mount 하위 섹션을 넣으면 그 아래
#   일반 속성(slurm.node_prefix 등)이 하위 섹션으로 흡수되어 노드 이름이
#   test-hpc-* → hpc-* 로 깨집니다(slurm↔CycleCloud 불일치).
#
# 반영: export_parameters → import_cluster --force 후,
#       대상 노드를 재기동(azslurm suspend/resume)하면 새로 뜬 노드에
#       디스크가 attach·(포맷)·마운트됩니다.
#       ※ 이미 실행 중인 노드에는 소급 적용되지 않습니다(재기동 필요).
#
# 두 가지 시나리오 중 하나를 선택하세요.
#   [A] 신규 디스크 생성   : 노드마다 새 관리 디스크를 만들어 붙임 (기본)
#   [B] 기존 디스크 마운트 : 이미 만들어 둔 관리 디스크를 VolumeId 로 붙임
#
# ⚠️ Persistent = true 는 여기(nodearraybase)에 쓸 수 없습니다.
#    nodearraybase 를 상속하는 것은 전부 [[nodearray ...]] 이고 이들은 VMSS 로 뜹니다.
#    VMSS 는 모든 인스턴스가 하나의 모델을 공유하므로 특정 디스크를 지목해 계속
#    붙여 쓰는 영속 볼륨을 표현할 수 없어, 노드 기동 시 다음 오류가 납니다.
#        Scaleset disk 'datadisk' cannot be persistent
#    영속 볼륨은 scheduler 나 [[node mynode]] 같은 **단일 노드([[node ...]])** 에만
#    지정합니다. 계산 노드에 데이터를 남겨야 한다면 디스크가 아니라 NFS 를 쓰세요.
#    (Extends = nodearraybase 를 쓰더라도 [[node ...]] 는 단독 VM 이라 영속이 됩니다.
#     갈림길은 상속이 아니라 node 냐 nodearray 냐 입니다.)
# =============================================================================


# =========================== [A] 신규 디스크 생성 + 마운트 ===================
#   Persistent 는 넣지 않습니다(기본 false). 이유는 상단 주석 참고.
#   노드마다 개별 디스크가 생성됩니다(공유 아님). 공유가 필요하면 NFS 사용.

        [[[volume datadisk]]]
        Size = 128
        SSD = True
        Mount = datadisk
        # StorageAccountType = Premium_LRS   # 선택: Standard_LRS / StandardSSD_LRS / Premium_LRS
        # Azure.LUN = 1                       # 선택: LUN 수동 지정

        [[[configuration cyclecloud.mounts.datadisk]]]
        mountpoint = /data
        fs_type = xfs                          # ext4 / xfs 등


# =========================== [B] 기존 디스크 attach + 마운트 =================
#   (위 [A] 대신 사용. 둘 다 넣지 마세요.)
#   관리 디스크는 동시에 한 노드에만 attach 되므로 **단독 [[node ...]] 전용**입니다.
#   nodearray(VMSS)에 넣으면 여러 노드가 같은 디스크를 붙이려 해 충돌하고,
#   Persistent = true 도 스케일셋에서는 거부됩니다([A] 상단 주석 참고).
#   VolumeId 는 대상 디스크의 전체 리소스 ID 로 교체하세요.

#       [[[volume existingdisk]]]
#       VolumeId = /subscriptions/<구독ID>/resourceGroups/<리소스그룹>/providers/Microsoft.Compute/disks/mydatadisk
#       Mount = existingdisk
#       Persistent = true
#
#       [[[configuration cyclecloud.mounts.existingdisk]]]
#       mountpoint = /data
#       fs_type = xfs
